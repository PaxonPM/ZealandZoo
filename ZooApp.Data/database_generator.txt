-- Create table Event
CREATE TABLE Event(
    event_id INT IDENTITY(1,1) PRIMARY KEY,
    title NVARCHAR(50) NOT NULL,
    description NVARCHAR(255) NOT NULL,
    start_time DATETIME NOT NULL,
    end_time DATETIME NOT NULL,
    location NVARCHAR(100) NOT NULL,
    max_participants INT NOT NULL CHECK (max_participants > 0),
    created_at DATETIME NOT NULL DEFAULT GETDATE(),

    CONSTRAINT CHK_Event_Time 
        CHECK (end_time > start_time)
);

-- Create table for Roles and Users
CREATE TABLE Role(
    role_id INT NOT NULL PRIMARY KEY,
    role_name NVARCHAR(20) NOT NULL
);


CREATE TABLE Users(
    id INT IDENTITY(1,1) NOT NULL PRIMARY KEY,
    name NVARCHAR(50) NULL,
    telefon NVARCHAR(8) NULL,
    email NVARCHAR(50) NULL,
    role_id INT NOT NULL,
    is_newsletter_member BIT NULL,
    pw_hash NVARCHAR(44) NULL,

    CONSTRAINT FK_Users_Role
        FOREIGN KEY (role_id)
        REFERENCES Role(role_id)
);


-- Create table for Event Participants
CREATE TABLE EventParticipants (
    event_id INT NOT NULL,
    user_id INT NOT NULL,
    joined_at DATETIME NOT NULL DEFAULT GETDATE(),

    PRIMARY KEY (event_id, user_id),

    CONSTRAINT FK_EventParticipants_Event
        FOREIGN KEY (event_id)
        REFERENCES Event(event_id)
        ON DELETE CASCADE,

    CONSTRAINT FK_EventParticipants_User
        FOREIGN KEY (user_id)
        REFERENCES Users(id)
);
GO

-- Create table for Item Categories
CREATE TABLE ItemCategories (
    id INT PRIMARY KEY IDENTITY,
    Name VARCHAR(50) NOT NULL
);

-- Create table for Inventory Items
CREATE TABLE InventoryItems (
    id INT PRIMARY KEY IDENTITY,
    category_id INT NOT NULL,
    Name VARCHAR(50) NOT NULL,
    quantity INT NOT NULL,
    CONSTRAINT UQ_InventoryItems_Name UNIQUE (Name),
    CONSTRAINT FK_InventoryItems_Category FOREIGN KEY (category_id) 
        REFERENCES ItemCategories(id)
);

-- Create table for Roles and Users
CREATE TABLE Role(
    role_id INT NOT NULL PRIMARY KEY,
    role_name NVARCHAR(20) NOT NULL
);

    is_newsletter_member BIT NULL,
